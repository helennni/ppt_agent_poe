📋 首次设置指南

  1. 前往 supabase.com 创建免费项目
  2. 在 Project Settings → API 中复制 URL 和 anon key
  3. 在 SQL Editor 中运行以下 SQL 创建数据表：
-- 创建 PPT 生成历史表
CREATE TABLE ppt_history (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  topic TEXT NOT NULL,
  style TEXT,
  language TEXT,
  slide_count TEXT,
  extra_req TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 启用行级安全
ALTER TABLE ppt_history ENABLE ROW LEVEL SECURITY;

-- 用户只能查看/插入自己的记录
CREATE POLICY "select_own" ON ppt_history
  FOR SELECT USING (auth.uid() = user_id);
CREATE POLICY "insert_own" ON ppt_history
  FOR INSERT WITH CHECK (auth.uid() = user_id);
CREATE POLICY "delete_own" ON ppt_history
  FOR DELETE USING (auth.uid() = user_id);
  1. 在 Authentication → Users 中添加用户（或开启注册）
  2. 将信息填入上方表单并保存
